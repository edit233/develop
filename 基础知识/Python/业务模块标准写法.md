## 概述
业务模块统一分层：**Router → Service → Repository** + models/schemas/exceptions。本大类给出六文件标准完整模板，参数处中文注明，可直接复制改造。

## 分层架构总览
目录结构：
```
modules/模块名/
├── models.py       # ORM 表模型（表结构映射）
├── schemas.py      # 请求/响应 Pydantic 模型
├── repository.py   # 数据访问层（只写 SQLAlchemy 查询）
├── service.py      # 业务层（事务边界、业务规则、抛异常）
├── router.py       # HTTP 层（参数解析、依赖注入）
└── exceptions.py   # 业务异常（可选）
```

依赖注入链：`get_session`（基础设施）→ `get_模块_service`（router 工厂函数）→ 接口函数。

## 六文件标准模板

### models.py —— ORM 表模型
```python
from sqlalchemy.orm import Mapped, mapped_column

from 公共models import Base, CreateAtMixin, UpdateAtMixin  # 见 SQLAlchemy 数据访问


class 实体名(Base, CreateAtMixin, UpdateAtMixin):
    """业务实体"""
    __tablename__ = "表名"  # 数据库表名

    id: Mapped[int] = mapped_column(primary_key=True)  # 主键
    名称: Mapped[str] = mapped_column(String(100))       # 字段说明
    状态: Mapped[str] = mapped_column(
        String(30), default="active", server_default="active",  # 双默认值
    )
```

### schemas.py —— 请求/响应模型
```python
from pydantic import Field, field_validator

from 公共schemas import BaseSchema  # 见 Pydantic 配置与模型


class 创建请求(BaseSchema):
    标题: str = Field(default="新标题", min_length=1, max_length=200)  # 标题约束

    @field_validator("标题", mode="before")  # 类型校验前清洗
    @classmethod
    def 清洗(cls, value: str) -> str:
        return value.strip()  # 去除首尾空格


class 响应模型(BaseSchema):
    id: int                    # 主键
    标题: str                   # 标题
    创建时间: datetime           # 由 ORM 转换（from_attributes）
```

### repository.py —— 数据访问层
```python
from sqlalchemy import func, or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from .models import 实体名


class 实体名Repository:
    def __init__(self, session: AsyncSession):  # 注入会话
        self.session = session

    async def find_page(self, 分类=None, 关键词=None, page=1, page_size=20):
        """分页查询：条件组装 + count + offset/limit（详见 SQLAlchemy 数据访问）"""
        conditions = [实体名.status == "active"]  # 默认条件
        if 分类:
            conditions.append(实体名.分类 == 分类)  # 可选精确过滤
        if 关键词:
            kw = f"%{关键词.strip()}%"
            conditions.append(or_(实体名.名称.ilike(kw)))  # 模糊搜索
        total = await self.session.scalar(
            select(func.count(实体名.id)).where(*conditions)
        ) or 0
        items = await self.session.scalars(
            select(实体名).where(*conditions)
            .order_by(实体名.id.asc())                    # 排序字段
            .offset((page - 1) * page_size).limit(page_size)
        )
        return items.all(), total

    async def find_owned(self, 对象_id, 用户_id) -> 实体名 | None:
        """存在性 + 归属校验：防越权访问"""
        return await self.session.scalar(
            select(实体名).where(
                实体名.id == 对象_id,
                实体名.user_id == 用户_id,  # 归属字段
            )
        )
```

### service.py —— 业务层
```python
from sqlalchemy.ext.asyncio import AsyncSession

from .exceptions import 资源NotFoundError
from .models import 实体名
from .repository import 实体名Repository


class 实体名Service:
    def __init__(self, session: AsyncSession):  # 注入会话，组装 Repository
        self.session = session
        self.repository = 实体名Repository(session)

    async def add(self, 用户_id: int, 标题: str) -> 实体名:
        async with self.session.begin():        # 事务：成功 commit，异常回滚
            对象 = 实体名(user_id=用户_id, 标题=标题)
            self.repository.add(对象)
            return 对象

    async def rename(self, 对象_id, 用户_id: int, 新标题: str) -> 实体名:
        async with self.session.begin():
            对象 = await self.repository.find_owned(对象_id, 用户_id)
            if 对象 is None:
                raise 资源NotFoundError        # 业务异常
            对象.标题 = 新标题
            await self.session.flush()         # 先落库，触发数据库回填
            await self.session.refresh(对象)   # 刷新拿回回填字段（时间戳等）
            return 对象
```

### router.py —— HTTP 层
```python
from typing import Annotated

from fastapi import APIRouter, Depends, Header, Query
from sqlalchemy.ext.asyncio import AsyncSession

from 公共schemas import PageResult  # 见 Pydantic 配置与模型
from 基础设施 import get_session

router = APIRouter(prefix="/api/v1/资源名", tags=["分组标签"])  # URL 前缀与文档分组


async def get_实体名_service(session: AsyncSession = Depends(get_session)):
    """依赖工厂：组装 Service 供接口注入"""
    return 实体名Service(session)


@router.get("", response_model=list[响应模型])
async def 列表接口(
    用户_id: Annotated[int, Header(alias="x-user-id")],  # 用户身份从请求头取
    service: 实体名Service = Depends(get_实体名_service),
):
    return await service.select(用户_id)


@router.get("/page", response_model=PageResult[响应模型])
async def 分页接口(
    分类: str | None = None,                           # 可选过滤参数
    page: int = Query(default=1, ge=1),                # 页码，最小 1
    page_size: int = Query(default=20, ge=1, le=100),  # 每页条数，1~100
    service: 实体名Service = Depends(get_实体名_service),
):
    return await service.list_xxx(分类, page, page_size)
```

### exceptions.py —— 业务异常
```python
from 公共异常 import ApplicationError  # 见 日志与异常处理


class 资源NotFoundError(ApplicationError):
    """资源不存在或不属于当前用户"""
    status_code = 400
    code = "RESOURCE_NOT_FOUND"
    message = "资源不存在或不属于当前用户"
```

## 核心模式
1. **分页模式**：默认条件 + 可选条件追加到同一 `conditions` 列表，count 与查询共用保证一致。
2. **归属校验模式**：`id == 对象_id AND user_id == 用户_id` 一条查询同时完成存在性与权限校验。
3. **事务模式**：`async with self.session.begin()` 包裹所有写操作，异常自动回滚。
4. **清洗校验模式**：`field_validator(mode="before")` 在类型校验前做 strip / None 归一化。
5. **依赖注入模式**：工厂函数 `get_实体名_service(Depends(get_session))` 统一组装 Service。
